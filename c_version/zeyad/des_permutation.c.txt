1#include "des_permutation.h"

// Standard DES IP Table (1-indexed from MSB)
static const uint8_t IP_TABLE[64] = {
    58, 50, 42, 34, 26, 18, 10, 2,
    60, 52, 44, 36, 28, 20, 12, 4,
    62, 54, 46, 38, 30, 22, 14, 6,
    64, 56, 48, 40, 32, 24, 16, 8,
    57, 49, 41, 33, 25, 17,  9, 1,
    59, 51, 43, 35, 27, 19, 11, 3,
    61, 53, 45, 37, 29, 21, 13, 5,
    63, 55, 47, 39, 31, 23, 15, 7
};

// Standard DES IP^-1 Table (1-indexed from MSB)
static const uint8_t FP_TABLE[64] = {
    40,  8, 48, 16, 56, 24, 64, 32,
    39,  7, 47, 15, 55, 23, 63, 31,
    38,  6, 46, 14, 54, 22, 62, 30,
    37,  5, 45, 13, 53, 21, 61, 29,
    36,  4, 44, 12, 52, 20, 60, 28,
    35,  3, 43, 11, 51, 19, 59, 27,
    34,  2, 42, 10, 50, 18, 58, 26,
    33,  1, 41,  9, 49, 17, 57, 25
};

// Standard DES E-Bit Selection Table (1-indexed from MSB of 32-bit input)
static const uint8_t E_TABLE[48] = {
    32,  1,  2,  3,  4,  5,
     4,  5,  6,  7,  8,  9,
     8,  9, 10, 11, 12, 13,
    12, 13, 14, 15, 16, 17,
    16, 17, 18, 19, 20, 21,
    20, 21, 22, 23, 24, 25,
    24, 25, 26, 27, 28, 29,
    28, 29, 30, 31, 32,  1
};

// Standard DES Permutation P Table (1-indexed from MSB of 32-bit input)
static const uint8_t P_TABLE[32] = {
    16,  7, 20, 21,
    29, 12, 28, 17,
     1, 15, 23, 26,
     5, 18, 31, 10,
     2,  8, 24, 14,
    32, 27,  3,  9,
    19, 13, 30,  6,
    22, 11,  4, 25
};

uint64_t des_initial_permutation(uint64_t in_block) {
    uint64_t out = 0;
    for (int i = 0; i < 64; i++) {
        uint8_t bit_pos = IP_TABLE[i]; // 1..64
        uint64_t bit = (in_block >> (64 - bit_pos)) & 1ULL;
        out = (out << 1) | bit;
    }
    return out;
}

uint64_t des_inverse_initial_permutation(uint64_t in_block) {
    uint64_t out = 0;
    for (int i = 0; i < 64; i++) {
        uint8_t bit_pos = FP_TABLE[i]; // 1..64
        uint64_t bit = (in_block >> (64 - bit_pos)) & 1ULL;
        out = (out << 1) | bit;
    }
    return out;
}

uint64_t des_expansion_permutation(uint32_t in_r) {
    uint64_t out = 0;
    for (int i = 0; i < 48; i++) {
        uint8_t bit_pos = E_TABLE[i]; // 1..32
        uint64_t bit = ((uint64_t)in_r >> (32 - bit_pos)) & 1ULL;
        out = (out << 1) | bit;
    }
    return out;
}

uint32_t des_p_permutation(uint32_t in_p) {
    uint32_t out = 0;
    for (int i = 0; i < 32; i++) {
        uint8_t bit_pos = P_TABLE[i]; // 1..32
        uint32_t bit = (in_p >> (32 - bit_pos)) & 1U;
        out = (out << 1) | bit;
    }
    return out;
}
